<script>


const map = L.map('map').setView([45.4642, 9.19], 8);


L.tileLayer(

  'https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png',

  {

    attribution: '&copy; OpenStreetMap'

  }

).addTo(map);


const socket = io();


const markers = {};


async function loadPositions() {


  const response =

    await fetch("/api/positions");


  const positions =

    await response.json();


  positions.forEach(updateMarker);

}


function updateSidebar(data) {


  const container =

    document.getElementById("targets");


  let html = "";


  Object.values(markers).forEach((m) => {


    html += `

      <div class="target">

        <b>${m.token}</b><br>

        ${m.timestamp}

      </div>

    `;

  });


  container.innerHTML = html;

}


function updateMarker(data) {


  const key = data.token;


  if (markers[key]) {


    markers[key].marker

      .setLatLng([data.lat, data.lon]);


    markers[key].timestamp =

      data.timestamp;


  } else {


    const marker = L.marker([

      data.lat,

      data.lon

    ])

    .addTo(map)

    .bindPopup(

      `

      <b>${data.token}</b><br>

      Accuratezza:

      ${Math.round(data.accuracy)} m

      `

    );


    markers[key] = {

      marker,

      token: data.token,

      timestamp: data.timestamp

    };


  }


  updateSidebar(data);

}


socket.on("position", (data) => {


  updateMarker(data);


  map.setView([data.lat, data.lon], 15);


});


loadPositions();


</script>
